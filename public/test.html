🧪 Тест анализатора сайтов

Инструкция

1. Откройте основное приложение: index.html

2. Нажмите кнопку "🧪 Запустить автоматические тесты"

3. Дождитесь завершения тестов

4. Проверьте результаты для обоих сайтов

Быстрый тест (в консоли браузера)

Откройте консоль браузера (F12) и выполните:

// Тест 1: Проверка URL
const url1 = 'https://expomap.ru/conference/nedelya-bezopasnosti-techexpert/';
const url2 = 'https://nic-conf.ru/';
console.log('URL 1:', new URL(url1).hostname);
console.log('URL 2:', new URL(url2).hostname);

// Тест 2: Проверка структуры данных
const testData = {
    overallScore: 75,
    content: { score: 80, issues: [] },
    ux: { score: 70, issues: [] },
    seo: { score: 65, issues: [] },
    technical: { score: 85, issues: [] }
};
console.log('Test data:', testData);
console.log('Average:', (testData.content.score + testData.ux.score + testData.seo.score + testData.technical.score) / 4);
